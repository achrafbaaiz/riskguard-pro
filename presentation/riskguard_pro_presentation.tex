\section{Introduction et Problématique}
%=======================================================================

%=======================================================================
% SLIDE 3 - Introduction
%=======================================================================
\begin{frame}{Introduction}
  \begin{itemize}
    \item Le risque de défaillance des entreprises est une préoccupation
          majeure pour le secteur bancaire.
    \pause
    \item Les banques doivent évaluer la solidité financière de leurs
          clients avant d'accorder des crédits.
    \pause
    \item Les méthodes classiques reposent sur l'analyse manuelle des
          bilans, ce qui est \alert{lent et subjectif}.
    \pause
    \item L'intelligence artificielle offre une alternative \alert{plus
          rapide et plus objective}.
  \end{itemize}
\end{frame}

%=======================================================================
% SLIDE 4 - Problematique
%=======================================================================
\begin{frame}{Problématique}
  \begin{block}{}
    \centering
    \large
    \textbf{« Comment peut-on prédire automatiquement le niveau de
    risque de défaillance d'une entreprise à partir de ses indicateurs
    financiers ? »}
  \end{block}
  \vfill
  \begin{itemize}
    \item \textbf{Difficulté :} les cas de faillite sont très rares dans
          les données ($\sim$3\%).
    \pause
    \item \textbf{Besoin :} un outil fiable, interprétable et facile à
          utiliser.
  \end{itemize}
\end{frame}

%=======================================================================
\section{Objectifs}
%=======================================================================

%=======================================================================
% SLIDE 5 - Objectifs
%=======================================================================
\begin{frame}{Objectifs du projet}
  \begin{itemize}
    \item Développer un modèle de \textit{Machine Learning} capable de
          prédire le risque de faillite.
    \pause
    \item Créer une plateforme web permettant aux analystes d'utiliser ce
          modèle facilement.
    \pause
    \item Permettre l'analyse \textbf{individuelle} et \textbf{en lot
          (batch)} des entreprises.
    \pause
    \item Fournir des résultats clairs avec 4 niveaux de risque :
      \begin{itemize}
        \item[\textcolor{rggreen}{$\blacksquare$}] Faible
        \item[\textcolor{orange}{$\blacksquare$}] Modéré
        \item[\textcolor{red}{$\blacksquare$}] Élevé
        \item[\textcolor{rgred}{$\blacksquare$}] Critique
      \end{itemize}
  \end{itemize}
\end{frame}

\section{Prétraitement des données}
%=======================================================================

%=======================================================================
% SLIDE 6 - Presentation du dataset
%=======================================================================
\begin{frame}{Le jeu de données}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \begin{itemize}
        \item \textbf{Source :} dataset public de prédiction de faillite
              d'entreprises.
        \item Contient environ \textbf{6 800 entreprises}.
        \item \textbf{96 ratios financiers} par entreprise (rentabilité,
              liquidité, endettement, etc.).
        \item Variable cible « Bankrupt? » :
              \textbf{0} (saine) ou \textbf{1} (en faillite).
        \item \alert{Problème : seulement $\sim$3\% d'entreprises en
              faillite} (données très déséquilibrées).
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.9]
        % axes
        \draw[->] (0,0) -- (0,4.2);
        \draw[->] (0,0) -- (4.2,0);
        % barre saines 97%
        \fill[rggreen] (0.6,0) rectangle (1.6,3.88);
        \node[above] at (1.1,3.88) {\small 97\%};
        \node[below,align=center] at (1.1,0) {\scriptsize Saines};
        % barre faillite 3%

        \node[above] at (3.1,0.36) {\small 3\%};
        \node[below,align=center] at (3.1,0) {\scriptsize Faillite};

    \end{column}
  \end{columns}
\end{frame}

%=======================================================================
% SLIDE 7 - Pretraitement des donnees
%=======================================================================
\begin{frame}{Prétraitement des données}
  \textbf{Les étapes réalisées :}
  \begin{enumerate}
    \item \textbf{Nettoyage :} suppression des valeurs infinies et
          remplacement des valeurs manquantes par la médiane.
    \pause
    \item \textbf{Valeurs aberrantes :} limitation (\textit{capping}) des
          valeurs extrêmes.
    \pause
    \item \textbf{Sélection des variables :} parmi les 96 ratios, nous
          avons retenu les \alert{10 indicateurs les plus pertinents}.
    \pause
    \item Ces 10 variables ont été choisies selon leur \textbf{pouvoir
          discriminant} entre entreprises saines et en difficulté.
  \end{enumerate}
\end{frame}

%=======================================================================
% SLIDE 7bis - Liste des 10 variables retenues
%=======================================================================
\begin{frame}{Les 10 variables retenues}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{enumerate}
        \small
        \item Net Income to Total Assets
        \item Retained Earnings to Total Assets
        \item Persistent EPS in the Last Four Seasons
        \item Net worth / Assets
        \item Debt ratio \%
      \end{enumerate}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{enumerate}
        \small
        \setcounter{enumi}{5}
        \item Borrowing dependency
        \item Liability to Equity
        \item Total debt / Total net worth
        \item Current Ratio
        \item Continuous interest rate (after tax)
      \end{enumerate}
    \end{column}
  \end{columns}
  \vfill
  \begin{exampleblock}{}
    \centering
    \small Rentabilité \quad$\bullet$\quad Structure financière
    \quad$\bullet$\quad Liquidité \quad$\bullet$\quad Endettement
  \end{exampleblock}
\end{frame}

\section{Génération de données synthétiques}
%=======================================================================

%=======================================================================
% SLIDE 8 - Donnees synthetiques : Pourquoi ?
%=======================================================================
\begin{frame}{Pourquoi ajouter des données synthétiques ?}
  \begin{itemize}
    \item Avec seulement \textbf{3\%} de cas de faillite, le modèle
          n'apprend pas correctement à détecter les entreprises à risque.
    \pause
    \item Il a tendance à toujours prédire « saine », car c'est le cas
          majoritaire.
    \pause
  \end{itemize}
  \begin{alertblock}{Solution}
    Créer de nouvelles données artificielles pour \textbf{équilibrer les
    classes}.
  \end{alertblock}
  \pause
  \begin{itemize}
    \item Ces données sont générées \alert{uniquement à partir des
          données d'entraînement} (pas de fuite d'information).
  \end{itemize}
\end{frame}

%=======================================================================
% SLIDE 9 - Donnees synthetiques : Comment ?
%=======================================================================
\begin{frame}{Génération des données synthétiques}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \begin{itemize}
        \small
        \item \textbf{Méthode :} interpolation entre profils
              d'entreprises saines et en faillite.
        \item Création de profils \textbf{intermédiaires} en mélangeant
              progressivement les caractéristiques des deux groupes.
        \item Ajout d'un léger \textbf{bruit gaussien} pour rendre les
              données plus réalistes et variées.
        \item \textbf{Résultat :} 3 000 nouvelles observations ajoutées
              aux données d'entraînement.
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering
      \begin{tikzpicture}[scale=1]
        \node[circle,fill=rggreen,inner sep=3pt,label=below:{\scriptsize Saine}] (a) at (0,0) {};
        \node[circle,fill=rgred,inner sep=3pt,label=below:{\scriptsize Faillite}] (b) at (4,0) {};
        \draw[dashed,rggray] (a) -- (b);
        \node[circle,fill=orange,inner sep=2.5pt] (m) at (2,0) {};
        \node[above=2pt] at (m) {\scriptsize Profil interpolé};
        % bruit gaussien
        \draw[->,rgblue,thick] (2,0.55) .. controls (2.6,1.1) .. (2.9,0.7);
        \node[rgblue,right] at (2.9,0.85) {\scriptsize bruit gaussien};
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\section{Le modèle XGBoost}
%=======================================================================

%=======================================================================
% SLIDE 10 - XGBoost : Principe
%=======================================================================
\begin{frame}{Le modèle XGBoost}
  \begin{itemize}
    \small
    \item \textbf{XGBoost} = \textit{eXtreme Gradient Boosting}.
    \item C'est un \textbf{ensemble d'arbres de décision} qui travaillent
          en équipe.
    \item \textbf{Principe :} chaque nouvel arbre corrige les erreurs des
          précédents.
    \item Les arbres sont ajoutés un par un, chacun se concentrant sur les
          cas mal classés.
    \item L'un des algorithmes les plus performants pour les
          \textbf{données tabulaires}.
  \end{itemize}
  \vfill
  \centering
  \begin{tikzpicture}[
      node distance=8mm,
      arbre/.style={rectangle,rounded corners,draw=rgblue,fill=rgblue!12,
                    minimum width=1.3cm,minimum height=0.8cm,font=\scriptsize},
      err/.style={font=\scriptsize\itshape,rggray},
      fin/.style={rectangle,rounded corners,draw=rggreen,fill=rggreen!15,
                  minimum width=1.6cm,minimum height=0.8cm,font=\scriptsize}]
    \node[arbre] (t1) {Arbre 1};
    \node[arbre,right=of t1] (t2) {Arbre 2};
    \node[arbre,right=of t2] (t3) {Arbre 3};
    \node[right=6mm of t3,font=\scriptsize] (dots) {$\cdots$};
    \node[fin,right=6mm of dots] (pred) {Prédiction\\finale};
    \draw[-{Stealth}] (t1) -- node[above,err]{erreurs} (t2);
    \draw[-{Stealth}] (t2) -- node[above,err]{erreurs} (t3);
    \draw[-{Stealth}] (t3) -- (dots);
    \draw[-{Stealth}] (dots) -- (pred);
  \end{tikzpicture}
\end{frame}

%=======================================================================
% SLIDE 11 - Pourquoi XGBoost ?
%=======================================================================
\begin{frame}{Pourquoi avoir choisi XGBoost ?}
  \begin{itemize}
    \item Très \textbf{performant} sur les données structurées (tableaux
          financiers).
    \pause
    \item Gère bien le \textbf{déséquilibre des classes} grâce à un
          paramètre de pondération.
    \pause
    \item \textbf{Rapide} à l'entraînement et à la prédiction.
    \pause
    \item Fournit l'\textbf{importance des variables}
          (interprétabilité).
    \pause
    \item Intègre une \textbf{régularisation} pour éviter le
          sur-apprentissage.
    \pause
    \item Algorithme reconnu et largement utilisé dans l'\textbf{industrie
          bancaire}.
  \end{itemize}
\end{frame}

\section{Résultats}
%=======================================================================

%=======================================================================
% SLIDE 15 - Resultats
%=======================================================================
\begin{frame}{Résultats obtenus}
  \begin{itemize}
    \item Le modèle a été évalué sur des \textbf{données réelles
          uniquement} (aucune donnée synthétique dans le test).
    \item Métriques obtenues sur le jeu de test :
      \begin{itemize}
        \item \textbf{ROC-AUC :} capacité du modèle à distinguer les deux
              classes.
        \item \textbf{F1-Score :} équilibre entre précision et rappel.
        \item \textbf{Recall (rappel) :} capacité à détecter les vrais cas
              de faillite.
      \end{itemize}
  \end{itemize}
  \begin{exampleblock}{Métriques de performance}
    \centering
    ROC-AUC : \textbf{[à compléter]} \quad
    F1-Score : \textbf{[à compléter]} \quad
    Recall : \textbf{[à compléter]}
  \end{exampleblock}
  \begin{itemize}
    \item Le modèle détecte efficacement les entreprises à risque tout en
          \textbf{limitant les fausses alertes}.
    \item Variables les plus importantes : \textbf{Debt ratio},
          \textbf{Net Income to Total Assets}, \textbf{Current Ratio}.
  \end{itemize}
\end{frame}
