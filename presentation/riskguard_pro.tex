\begin{frame}{Contexte \& Problématique}
  \begin{itemize}

\section{Données synthétiques}
%==============================================================================
\begin{frame}{Pourquoi rééquilibrer~?}
  \begin{itemize}
    \item 3\,\% de positifs $\rightarrow$ le modèle prédit \textbf{toujours « sain »}.
    \item Accuracy 97\,\% \textbf{trompeuse} ; \textbf{recall} faillite $\approx$ 0.
    \item Solution : \textbf{augmenter} la classe minoritaire.
  \end{itemize}
  \vfill
  \centering
  \footnotesize Confusion : \emph{avant} \;
  \begin{tabular}{|c|c|}\hline 97 & 0\\\hline 3 & 0\\\hline\end{tabular}
  \quad $\rightarrow$ \quad \emph{après} \;
  \begin{tabular}{|c|c|}\hline 88 & 9\\\hline 1 & 12\\\hline\end{tabular}
\end{frame}

\begin{frame}{SMOTE + Bruit Gaussien}
  \begin{block}{SMOTE (interpolation $k$-NN)}
    \[ x_{\text{new}} = x_i + \lambda\,(x_{\text{voisin}} - x_i),\quad \lambda \in [0,1]. \]
  \end{block}
  \begin{block}{Bruit gaussien (variabilité réaliste)}
    \[ x_{\text{final}} = x_{\text{smote}} + \varepsilon,\quad
       \varepsilon \sim \mathcal{N}(0,\sigma^2),\; \sigma = 0.01\,\hat{\sigma}. \]
  \end{block}
  \centering {\footnotesize $\Rightarrow$ dataset équilibré, diversifié, anti-surapprentissage.}
\end{frame}

\begin{frame}{Pourquoi XGBoost~?}
  \begin{columns}[T]
    \column{0.55\textwidth}
      \begin{itemize}
        \item Meilleur \textbf{F1 / AUC}.
        \item Déséquilibre : \texttt{scale\_pos\_weight}.
        \item \textbf{Régularisation} L1+L2.
        \item Robuste aux outliers.
        \item \textbf{Feature importance} (interprétable).
        \item Inférence $<$ 100 ms.
      \end{itemize}
    \column{0.45\textwidth}
      \begin{exampleblock}{Écarté}
        \begin{itemize}
          \item Rég. log. : trop simple.
          \item SVM : lent, opaque.
          \item RF : sans boosting.
          \item DL : « boîte noire ».
        \end{itemize}
      \end{exampleblock}
  \end{columns}
  \vfill
  \centering {\footnotesize Tuning \textbf{Optuna} (bayésien, 30 essais) + CV stratifiée 5-fold.}
\end{frame}
