\section{Contexte \& Problématique}
%==============================================================================
\begin{frame}{Contexte \& Problématique}
  \begin{columns}[T]
    \column{0.55\textwidth}
      \begin{itemize}
        \item Le secteur bancaire marocain fait face à un \textbf{risque croissant
              de défaillance} d'entreprises.
        \item Les méthodes traditionnelles (scoring manuel, ratios simples) sont
              \textbf{limitées et subjectives}.
        \item Besoin d'un outil \textbf{automatisé, fiable et rapide} pour
              classifier le niveau de risque.
      \end{itemize}
      \vfill
      \begin{block}{Question centrale}
        Comment prédire la \emph{probabilité de faillite} d'une entreprise à
        partir de ses \emph{ratios financiers}~?

\section{Objectifs du projet}
%==============================================================================
\begin{frame}{Objectifs du projet}
  \begin{itemize}
    \item Construire un \textbf{pipeline ML complet} : de la donnée brute à la
          prédiction en temps réel.
    \item Classifier les entreprises en \textbf{4 niveaux de risque} : Faible,
          Modéré, Élevé, Critique.
    \item Développer une \textbf{interface web intuitive} pour les analystes
          bancaires.
    \item Fournir des explications sur les décisions du modèle
          (\textbf{interprétabilité via SHAP}).
    \item Permettre l'\textbf{analyse unitaire et batch} (CSV / Excel).
  \end{itemize}
\end{frame}

  \begin{columns}[T]
    \column{0.6\textwidth}

        \item Détection et suppression des \textbf{doublons}.
        \item Valeurs manquantes : \textbf{imputation par la médiane}

\section{Données synthétiques (SMOTE + Bruit Gaussien)}
%==============================================================================
\begin{frame}{Pourquoi générer des données synthétiques~?}
  \begin{columns}[T]
    \column{0.55\textwidth}
      \begin{itemize}
        \item Avec $\sim$3\,\% de cas positifs, le modèle apprend à
              \textbf{toujours prédire « sain »}.
        \item \textbf{Accuracy trompeuse} : 97\,\% en prédisant toujours 0
              $\rightarrow$ inutile.
        \item Le \textbf{recall} sur la classe minoritaire (faillite) est
              quasi nul sans rééquilibrage.
        \item \textbf{Solution} : augmenter la classe minoritaire avec des
              données synthétiques réalistes.
      \end{itemize}
    \column{0.45\textwidth}
      \centering
      \footnotesize Matrices de confusion\\[4pt]
      \begin{tabular}{c@{\hskip 1.2em}c}
        \emph{Avant} & \emph{Après}\\[2pt]
        \begin{tabular}{|c|c|}\hline 97 & 0\\\hline 3 & 0\\\hline\end{tabular} &
        \begin{tabular}{|c|c|}\hline 88 & 9\\\hline 1 & 12\\\hline\end{tabular}\\[6pt]
      \end{tabular}\\[4pt]
      {\scriptsize Lignes = réel (0/1), colonnes = prédit (0/1).}
  \end{columns}
\end{frame}

\begin{frame}{SMOTE + Bruit Gaussien}
  \begin{block}{Étape 1 --- SMOTE (Synthetic Minority Over-sampling)}
    Interpolation entre voisins proches ($k$-NN) : pour chaque exemple
    minoritaire, un point est généré sur le segment vers un voisin.
    \[ x_{\text{new}} = x_i + \lambda\,(x_{\text{voisin}} - x_i),
       \qquad \lambda \in [0,1]. \]
  \end{block}
  \begin{block}{Étape 2 --- Ajout de bruit gaussien}
    Un léger bruit normal évite des données trop « lisses » et limite le
    sur-apprentissage sur des patterns artificiellement réguliers.
    \[ x_{\text{final}} = x_{\text{smote}} + \varepsilon,
       \qquad \varepsilon \sim \mathcal{N}(0,\sigma^2),\;
       \sigma = 0.01 \times \text{écart-type}. \]
  \end{block}
\end{frame}

\begin{frame}{Transformation en 4 classes de risque}
  \centering

\section{Pourquoi XGBoost~?}
%==============================================================================
\begin{frame}{Pourquoi XGBoost~? --- Justification}
  \begin{columns}[T]
    \column{0.58\textwidth}
      \begin{enumerate}
        \item \textbf{Performance} : meilleur F1 et AUC-ROC.
        \item \textbf{Gestion du déséquilibre} : \texttt{scale\_pos\_weight}.
        \item \textbf{Régularisation L1 + L2} : anti-surapprentissage.
        \item \textbf{Robustesse aux outliers} (arbres de décision).
        \item \textbf{Feature importance} $\Rightarrow$ interprétabilité.
        \item \textbf{Optuna} : tuning bayésien (30 essais).
        \item \textbf{Inférence rapide} ($<$ 100 ms).
        \item \textbf{Écosystème mature}, standard en finance.
      \end{enumerate}
    \column{0.42\textwidth}
      \begin{exampleblock}{Pourquoi pas les autres~?}
        \begin{itemize}
          \item Rég. logistique : pas de non-linéarités.
          \item SVM : lent, peu interprétable.
          \item Random Forest : sans boosting séquentiel.
          \item Deep Learning : trop complexe, « boîte noire ».
        \end{itemize}
      \end{exampleblock}
      \centering
      \begin{tikzpicture}[node distance=0.7cm,
        tree/.style={circle,draw,minimum size=0.6cm,font=\scriptsize}]
        \node[tree](t1){$T_1$};
        \node[tree,right=of t1](t2){$T_2$};
        \node[tree,right=of t2](t3){$T_3$};
        \draw[-{Latex}](t1)--(t2);
        \draw[-{Latex}](t2)--(t3);
      \end{tikzpicture}\\
      {\scriptsize Gradient boosting : arbres séquentiels.}
  \end{columns}
\end{frame}

\begin{frame}{Optimisation des hyperparamètres avec Optuna}
  \begin{columns}[T]
    \column{0.5\textwidth}
      \begin{itemize}
        \item \textbf{Optimisation bayésienne} (plus intelligente que Grid Search).
        \item \textbf{30 essais} d'optimisation automatique.
        \item \textbf{Validation croisée stratifiée} (5-fold).
      \end{itemize}
      \begin{block}{Hyperparamètres optimisés}
